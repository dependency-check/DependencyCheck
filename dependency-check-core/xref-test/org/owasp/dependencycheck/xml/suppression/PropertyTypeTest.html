18  package org.owasp.dependencycheck.xml.suppression;
19  
20  import org.junit.jupiter.api.Test;
21  import org.owasp.dependencycheck.BaseTest;
22  
23  import static org.junit.jupiter.api.Assertions.assertEquals;
24  import static org.junit.jupiter.api.Assertions.assertFalse;
25  import static org.junit.jupiter.api.Assertions.assertTrue;
26  
27  /**
28   *
29   * @author Jeremy Long
30   */
31  class PropertyTypeTest extends BaseTest {
32  
33      /**
34       * Test of set and getValue method, of class PropertyType.
35       */
36      @Test
37      void testSetGetValue() {
38  
39          PropertyType instance = new PropertyType();
40          String expResult = "test";
41          instance.setValue(expResult);
42          String result = instance.getValue();
43          assertEquals(expResult, result);
44      }
45  
46      /**
47       * Test of isRegex method, of class PropertyType.
48       */
49      @Test
50      void testIsRegex() {
51          PropertyType instance = new PropertyType();
52          assertFalse(instance.isRegex());
53          instance.setRegex(true);
54          assertTrue(instance.isRegex());
55      }
56  
57      /**
58       * Test of isCaseSensitive method, of class PropertyType.
59       */
60      @Test
61      void testIsCaseSensitive() {
62          PropertyType instance = new PropertyType();
63          assertFalse(instance.isCaseSensitive());
64          instance.setCaseSensitive(true);
65          assertTrue(instance.isCaseSensitive());
66      }
67  
68      /**
69       * Test of matches method, of class PropertyType.
70       */
71      @Test
72      void testMatches() {
73          String text = "Simple";
74  
75          PropertyType instance = new PropertyType();
76          instance.setValue("simple");
77          assertTrue(instance.matches(text));
78          instance.setCaseSensitive(true);
79          assertFalse(instance.matches(text));
80  
81          instance.setValue("s.*le");
82          instance.setRegex(true);
83          assertFalse(instance.matches(text));
84          instance.setCaseSensitive(false);
85          assertTrue(instance.matches(text));
86      }
87  }
