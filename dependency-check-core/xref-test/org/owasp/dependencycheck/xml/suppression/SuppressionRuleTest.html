155     @Test
156     void testCpeMatches() throws Exception {
157         Identifier identifier = new CpeIdentifier("microsoft", ".net_framework", "4.5", Confidence.HIGHEST);
158 
159         SuppressionRule instance = new SuppressionRule();
160         boolean expResult = true;
161         boolean result = instance.identifierMatches(PropertyType.of("cpe:/a:microsoft:.net_framework:4.5"), identifier);
162         assertEquals(expResult, result);
163 
164         expResult = false;
165         result = instance.identifierMatches(PropertyType.of("cpe:/a:microsoft:.net_framework:4.0"), identifier);
166         assertEquals(expResult, result);
167 
168         expResult = false;
169         result = instance.identifierMatches(PropertyType.caseSensitive("CPE:/a:microsoft:.net_framework:4.5"), identifier);
170         assertEquals(expResult, result);
171 
172         expResult = true;
173         result = instance.identifierMatches(PropertyType.of("cpe:/a:microsoft:.net_framework"), identifier);
174         assertEquals(expResult, result);
175 
176         expResult = true;
177         result = instance.identifierMatches(PropertyType.regex("cpe:/a:microsoft:.*"), identifier);
178         assertEquals(expResult, result);
179 
180         expResult = false;
181         result = instance.identifierMatches(PropertyType.regexCaseSensitive("CPE:/a:microsoft:.*"), identifier);
182         assertEquals(expResult, result);
183 
184         expResult = false;
185         result = instance.identifierMatches(PropertyType.regex("cpe:/a:apache:.*"), identifier);
186         assertEquals(expResult, result);
187 
188         identifier = new CpeIdentifier("apache", "tomcat", "7.0", Confidence.HIGH);
189         expResult = true;
190         result = instance.identifierMatches(PropertyType.of("cpe:/a:apache:tomcat"), identifier);
191         assertEquals(expResult, result);
192 
193         identifier = new CpeIdentifier(CpeParser.parse("cpe:/a:apache:tomcat_subproduct"), Confidence.HIGH);
194         expResult = false;
195         result = instance.identifierMatches(PropertyType.of("cpe:/a:apache:tomcat:"), identifier);
196         assertEquals(expResult, result);
197 
198         identifier = new CpeIdentifier(CpeParser.parse("cpe:/a:apache:tomcat"), Confidence.HIGH);
199         expResult = true;
200         result = instance.identifierMatches(PropertyType.of("cpe:/a:apache:tomcat:"), identifier);
201         assertEquals(expResult, result);
202     }

217      * Test of process method, of class SuppressionRule.
218      */
219     @Test
220     void testProcess() throws CpeValidationException {
221         File struts = BaseTest.getResourceAsFile(this, "maven-lib/struts2-core-2.1.2.jar");
222         Dependency dependency = new Dependency(struts);
223         CpeIdentifier cpeId = new CpeIdentifier("microsoft", ".net_framework", "4.5", Confidence.HIGH);
224         dependency.addVulnerableSoftwareIdentifier(cpeId);
225         String sha1 = dependency.getSha1sum();
226         dependency.setSha1sum("384FAA82E193D4E4B0546059CA09572654BC3970");
227         Vulnerability v = createVulnerability();
228         dependency.addVulnerability(v);
229 
230         //cwe
231         SuppressionRule instance = new SuppressionRule();
232         instance.setSha1(sha1);
233         instance.addCwe("287");
234         instance.process(dependency);
235         assertEquals(1, dependency.getVulnerabilities().size());
236         dependency.setSha1sum(sha1);
237         instance.process(dependency);
238         assertTrue(dependency.getVulnerabilities().isEmpty());
239         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
240 
241         //cvss
242         dependency.addVulnerability(v);
243         instance = new SuppressionRule();
244         instance.addCvssBelow(5.0);
245         instance.process(dependency);
246         assertEquals(1, dependency.getVulnerabilities().size());
247         instance.addCvssBelow(8.0);
248         instance.process(dependency);
249         assertTrue(dependency.getVulnerabilities().isEmpty());
250         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
251 
252         //cve
253         dependency.addVulnerability(v);
254         instance = new SuppressionRule();
255         instance.addCve("CVE-2012-1337");
256         instance.process(dependency);
257         assertEquals(1, dependency.getVulnerabilities().size());
258         instance.addCve("CVE-2013-1337");
259         instance.process(dependency);
260         assertTrue(dependency.getVulnerabilities().isEmpty());
261         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
262 
263         //cpe
264         instance = new SuppressionRule();
265         instance.addCpe(PropertyType.of("cpe:/a:microsoft:.net_framework:4.0"));
266         instance.process(dependency);
267         assertEquals(1, dependency.getVulnerableSoftwareIdentifiers().size());
268         instance.addCpe(PropertyType.of("cpe:/a:microsoft:.net_framework:4.5"));
269         instance.setFilePath(PropertyType.regex(".*"));
270         instance.process(dependency);
271         assertTrue(dependency.getVulnerableSoftwareIdentifiers().isEmpty());
272         assertEquals(1, dependency.getSuppressedIdentifiers().size());
273 
274         instance = new SuppressionRule();
275 
276         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "4.0", Confidence.HIGH));
277         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "4.5", Confidence.HIGH));
278         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "5.0", Confidence.HIGH));
279         instance.addCpe(PropertyType.of("cpe:/a:microsoft:.net_framework"));
280         instance.setBase(true);
281         assertEquals(3, dependency.getVulnerableSoftwareIdentifiers().size());
282         assertEquals(1, dependency.getSuppressedIdentifiers().size());
283         instance.process(dependency);
284         assertTrue(dependency.getVulnerableSoftwareIdentifiers().isEmpty());
285         assertEquals(1, dependency.getSuppressedIdentifiers().size());
286     }

337     @ParameterizedTest
338     @CsvSource(
339             value = {
340                     // vuln has cvss cvssV2=6.0, cvssV3=8.0, cvssV4 is not set
341                     // cvssV2Below, cvssV3Below, isSuppressed
342                     "   5.0,  7.0,  false",
343                     "   7.0,  7.0,  false",
344                     "  null,  7.0,  false",
345                     "   5.0,  9.0,  false",
346                     "   7.0,  9.0,  true",
347                     "  null,  9.0,  true",
348                     "   5.0, null,  false",
349                     "   7.0, null,  true",
350                     "  null, null,  false",
351 
352                     "   6.0,  8.0,  false", // cvssVnBelows are exclusive
353             },
354             nullValues = "null")
355     void testVersionSpecificThresholding(Double cvssV2Below, Double cvssV3Below, boolean isSuppressed) {
356         Dependency dependency = createDependencyWithDifferentScores();
357         SuppressionRule rule = new SuppressionRule();
358 
359         if (cvssV2Below != null) {
360             rule.addCvssV2Below(cvssV2Below);
361         }
362         if (cvssV3Below != null) {
363             rule.addCvssV3Below(cvssV3Below);
364         }
365         rule.process(dependency);
366 
367         assertEquals(isSuppressed ? 0 : 1, dependency.getVulnerabilities().size(),
368                 String.format("cvssV2Below=%s, cvssV3Below=%s: expecting vulnerability to be %s",
369                         cvssV2Below == null ? "not set" : String.format("%.1f", cvssV2Below),
370                         cvssV3Below == null ? "not set" : String.format("%.1f", cvssV3Below),
371                         isSuppressed ? "suppressed" : "not suppressed")
372         );
373         assertEquals(isSuppressed ? 1 : 0, dependency.getSuppressedVulnerabilities().size());
374     }

378     void testMismatchOfThresholdAndAvailableCVEVersion() {
379         // vuln with only a cvss v2 score
380         File spring = BaseTest.getResourceAsFile(this, "maven-lib/spring-security-web-3.0.0.RELEASE.jar");
381         Dependency dependency = new Dependency(spring);
382         Vulnerability v = new Vulnerability();
383         CvssV2 cvss = CvssUtil.vectorToCvssV2("/AV:N/AC:L/Au:N/C:P/I:P/A:P", 6.0);
384         v.setCvssV2(cvss);
385         dependency.addVulnerability(v);
386 
387         // rule with only a V3 threshold
388         SuppressionRule rule = new SuppressionRule();
389         rule.addCvssV3Below(7.0);
390 
391         assertEquals(1, dependency.getVulnerabilities().size(),
392                 "Since threshold and score versions are different versions the vuln should not be suppressed");
393     }

397         Dependency dependency = createDependencyWithDifferentScores();
398         SuppressionRule rule = new SuppressionRule();

409         Dependency dependency = createDependencyWithDifferentScores();

433     private Dependency createDependencyWithDifferentScores() {
434         File spring = BaseTest.getResourceAsFile(this, "maven-lib/spring-security-web-3.0.0.RELEASE.jar");
435         Dependency dependency = new Dependency(spring);
436         Vulnerability v = new Vulnerability();
437         CvssV2 cvss = CvssUtil.vectorToCvssV2("/AV:N/AC:L/Au:N/C:P/I:P/A:P", 6.0);
438         v.setCvssV2(cvss);
439         CvssV3 cvss3 = CvssUtil.vectorToCvssV3("CVSS:3.1/AV:N/AC:L/PR:N/UI:N/S:U/C:H/I:H/A:H", 8.0);
440         v.setCvssV3(cvss3);
441         dependency.addVulnerability(v);
442         return dependency;
443     }
