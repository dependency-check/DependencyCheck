415      * Test of process method, of class SuppressionRule.

417     @Test
418     void testProcess() throws CpeValidationException {
419         //File struts = new File(this.getClass().getClassLoader().getResource("struts2-core-2.1.2.jar").getPath());
420         File struts = BaseTest.getResourceAsFile(this, "struts2-core-2.1.2.jar");
421         Dependency dependency = new Dependency(struts);
422         CpeIdentifier cpeId = new CpeIdentifier("microsoft", ".net_framework", "4.5", Confidence.HIGH);
423         dependency.addVulnerableSoftwareIdentifier(cpeId);
424         String sha1 = dependency.getSha1sum();
425         dependency.setSha1sum("384FAA82E193D4E4B0546059CA09572654BC3970");
426         Vulnerability v = createVulnerability();
427         dependency.addVulnerability(v);
428 
429         //cwe
430         SuppressionRule instance = new SuppressionRule();
431         instance.setSha1(sha1);
432         instance.addCwe("287");
433         instance.process(dependency);
434         assertEquals(1, dependency.getVulnerabilities().size());
435         dependency.setSha1sum(sha1);
436         instance.process(dependency);
437         assertTrue(dependency.getVulnerabilities().isEmpty());
438         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
439 
440         //cvss
441         dependency.addVulnerability(v);
442         instance = new SuppressionRule();
443         instance.addCvssBelow(5.0);
444         instance.process(dependency);
445         assertEquals(1, dependency.getVulnerabilities().size());
446         instance.addCvssBelow(8.0);
447         instance.process(dependency);
448         assertTrue(dependency.getVulnerabilities().isEmpty());
449         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
450 
451         //cve
452         dependency.addVulnerability(v);
453         instance = new SuppressionRule();
454         instance.addCve("CVE-2012-1337");
455         instance.process(dependency);
456         assertEquals(1, dependency.getVulnerabilities().size());
457         instance.addCve("CVE-2013-1337");
458         instance.process(dependency);
459         assertTrue(dependency.getVulnerabilities().isEmpty());
460         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
461 
462         //cpe
463         instance = new SuppressionRule();
464         PropertyType pt = new PropertyType();
465         pt.setValue("cpe:/a:microsoft:.net_framework:4.0");
466         instance.addCpe(pt);
467         instance.process(dependency);
468         assertEquals(1, dependency.getVulnerableSoftwareIdentifiers().size());
469         pt = new PropertyType();
470         pt.setValue("cpe:/a:microsoft:.net_framework:4.5");
471         instance.addCpe(pt);
472         pt = new PropertyType();
473         pt.setValue(".*");
474         pt.setRegex(true);
475         instance.setFilePath(pt);
476         instance.process(dependency);
477         assertTrue(dependency.getVulnerableSoftwareIdentifiers().isEmpty());
478         assertEquals(1, dependency.getSuppressedIdentifiers().size());
479 
480         instance = new SuppressionRule();
481 
482         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "4.0", Confidence.HIGH));
483         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "4.5", Confidence.HIGH));
484         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("microsoft", ".net_framework", "5.0", Confidence.HIGH));
485         pt = new PropertyType();
486         pt.setValue("cpe:/a:microsoft:.net_framework");
487         instance.addCpe(pt);
488         instance.setBase(true);
489         assertEquals(3, dependency.getVulnerableSoftwareIdentifiers().size());
490         assertEquals(1, dependency.getSuppressedIdentifiers().size());
491         instance.process(dependency);
492         assertTrue(dependency.getVulnerableSoftwareIdentifiers().isEmpty());
493         assertEquals(1, dependency.getSuppressedIdentifiers().size());
494     }

534     @Test
535     void testProcessVulnerabilityNames() throws CpeValidationException, MalformedPackageURLException {
536         File spring = BaseTest.getResourceAsFile(this, "spring-security-web-3.0.0.RELEASE.jar");
537         Dependency dependency = new Dependency(spring);
538         dependency.addVulnerableSoftwareIdentifier(new CpeIdentifier("vmware", "springsource_spring_security", "3.0.0", Confidence.HIGH));
539         dependency.addSoftwareIdentifier(new PurlIdentifier("maven", "org.springframework.security", "spring-security-web", "3.0.0.RELEASE", Confidence.HIGH));
540 
541         dependency.addVulnerability(createVulnerability());
542         SuppressionRule instance = new SuppressionRule();
543         //gav
544         PropertyType pt = new PropertyType();
545         pt.setValue("pkg:maven/org.springframework.security/spring-security-web@3.0.0.RELEASE");
546         pt.setRegex(false);
547         pt.setCaseSensitive(false);
548         instance.setPackageUrl(pt);
549 
550         pt = new PropertyType();
551         pt.setValue("CVE-2013-1338");
552         instance.addVulnerabilityName(pt);
553 
554         instance.process(dependency);
555         assertEquals(1, dependency.getVulnerabilities().size());
556         assertEquals(0, dependency.getSuppressedVulnerabilities().size());
557 
558 
559         pt = new PropertyType();
560         pt.setValue("CVE-2013-1337");
561         instance.addVulnerabilityName(pt);
562 
563         instance.process(dependency);
564         assertEquals(0, dependency.getVulnerabilities().size());
565         assertEquals(1, dependency.getSuppressedVulnerabilities().size());
566     }

568     private Vulnerability createVulnerability() {
569         Vulnerability v = new Vulnerability();
570         v.addCwe("CWE-287 Improper Authentication");
571         v.setName("CVE-2013-1337");
572 
573         CvssV2 cvss = CvssUtil.vectorToCvssV2("/AV:N/AC:L/Au:N/C:P/I:P/A:P", 7.5);
574         v.setCvssV2(cvss);
575         return v;
576     }
