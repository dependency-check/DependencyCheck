20  import org.junit.jupiter.api.Test;
21  import org.owasp.dependencycheck.BaseTest;
22  import org.owasp.dependencycheck.dependency.Dependency;
23  import org.owasp.dependencycheck.dependency.EvidenceType;
24  
25  import java.io.File;
26  
27  import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
28  import static org.junit.jupiter.api.Assertions.assertEquals;
29  import static org.junit.jupiter.api.Assertions.assertTrue;
30  
31  /**
32   *
33   * @author Jeremy Long
34   */
35  class FileNameAnalyzerTest extends BaseTest {
36  
37      /**
38       * Test of getName method, of class FileNameAnalyzer.
39       */
40      @Test
41      void testGetName() {
42          FileNameAnalyzer instance = new FileNameAnalyzer();
43          String expResult = "File Name Analyzer";
44          String result = instance.getName();
45          assertEquals(expResult, result);
46      }
47  
48      /**
49       * Test of getAnalysisPhase method, of class FileNameAnalyzer.
50       */
51      @Test
52      void testGetAnalysisPhase() {
53          FileNameAnalyzer instance = new FileNameAnalyzer();
54          AnalysisPhase expResult = AnalysisPhase.INFORMATION_COLLECTION;
55          AnalysisPhase result = instance.getAnalysisPhase();
56          assertEquals(expResult, result);
57      }
58  
59      /**
60       * Test of analyze method, of class FileNameAnalyzer.
61       */
62      @Test
63      void testAnalyze() throws Exception {
64          //File struts = new File(this.getClass().getClassLoader().getResource("struts2-core-2.1.2.jar").getPath());
65          File struts = BaseTest.getResourceAsFile(this, "struts2-core-2.1.2.jar");
66          Dependency resultStruts = new Dependency(struts);
67          //File axis = new File(this.getClass().getClassLoader().getResource("axis2-adb-1.4.1.jar").getPath());
68          File axis = BaseTest.getResourceAsFile(this, "axis2-adb-1.4.1.jar");
69          Dependency resultAxis = new Dependency(axis);
70          FileNameAnalyzer instance = new FileNameAnalyzer();
71          instance.analyze(resultStruts, null);
72          assertTrue(resultStruts.getEvidence(EvidenceType.VENDOR).toString().toLowerCase().contains("struts"));
73  
74          instance.analyze(resultAxis, null);
75          assertTrue(resultStruts.getEvidence(EvidenceType.VERSION).toString().toLowerCase().contains("2.1.2"));
76  
77      }
78  
79      /**
80       * Test of prepare method, of class FileNameAnalyzer.
81       */
82      @Test
83      void testInitialize() {
84          FileNameAnalyzer instance = new FileNameAnalyzer();
85          assertDoesNotThrow(() -> {
86              instance.initialize(getSettings());
87              instance.prepare(null);
88          });
89          assertTrue(instance.isEnabled());
90      }
91  
92      /**
93       * Test of close method, of class FileNameAnalyzer.
94       */
95      @Test
96      void testClose() {
97          FileNameAnalyzer instance = new FileNameAnalyzer();
98          assertDoesNotThrow(instance::close);
99      }
100 }
