94          @ParameterizedTest
95          @EnumSource(value = OssIndexAnalyzer.OssIndexKnownError.class)
96          void should_return_a_dedicated_error_messages_for_responses_where_possible(OssIndexAnalyzer.OssIndexKnownError knownError) throws Exception {
97              // Given
98              Settings settings = getSettings();
99              setSonatypeGuideCredentials(settings);
100             settings.setBoolean(KEYS.ANALYZER_OSSINDEX_USE_CACHE, false);
101 
102             Engine engine = new Engine(settings);
103             analyzer.initialize(settings);
104             analyzer.prepareAnalyzer(engine);
105             Dependency dependency = addTestDependencyTo(engine);
106 
107             // When
108             try (engine; var clientProvider = withClientCreation(throwingOssIndex(new Transport.TransportException("Unexpected response; status: " + knownError.statusCode)))) {
109                 Throwable e = assertThrows(AnalysisException.class, () -> analyzer.analyzeDependency(dependency, engine));
110                 assertThat(e.getMessage(), containsString("Sonatype OSS Index / Guide " + knownError.userMessage));
111 
112                 if (!knownError.fatal) {
113                     assertTrue(analyzer.isEnabled());
114                 } else {
115                     clientProvider.clearInvocations();
116                     assertDoesNotThrow(() -> analyzer.analyzeDependency(dependency, engine),
117                             "Analysis exception thrown but should have been a no-op from earlier fatal error");
118                     clientProvider.verifyNoInteractions();
119                     assertFalse(analyzer.isEnabled());
120                 }
121 
122                 analyzer.setEnabled(true);
123                 settings.setBoolean(Settings.KEYS.ANALYZER_OSSINDEX_WARN_ONLY_ON_REMOTE_ERRORS, true);
124                 analyzer.initialize(settings);
125                 assertDoesNotThrow(() -> analyzer.analyzeDependency(dependency, engine),
126                         "Analysis exception thrown upon remote error although only a warning should have been logged");
127                 assertThat(analyzer.isEnabled(), is(!knownError.fatal));
128             }
129         }

159         @SuppressWarnings("resource")
160         void should_retry_with_delay_non_fatal_errors() throws Exception {
161             // Given
162             analyzer = spy(new OssIndexAnalyzer());
163

165             setSonatypeGuideCredentials(settings);
166             Engine engine = new Engine(settings);
167

169             analyzer.prepareAnalyzer(engine);
170 
171             Dependency dependency = addTestDependencyTo(engine);
172 
173             // When
174             SocketTimeoutException nonFatalError = new SocketTimeoutException("Read timed out");
175             try (engine; var clientProvider = withClientCreation(throwingOssIndex(nonFatalError))) {
176                 assertThrows(AnalysisException.class, () -> analyzer.analyzeDependency(dependency, engine));
177                 clientProvider.verify(() -> OssIndexClientProvider.create(settings));
178                 assertTrue(analyzer.isEnabled());
179 
180                 // Retry with no delay
181                 clientProvider.clearInvocations();
182                 assertThrows(AnalysisException.class, () -> analyzer.analyzeDependency(dependency, engine));
183                 assertTrue(analyzer.isEnabled());
184                 clientProvider.verify(() -> OssIndexClientProvider.create(settings));
185                 verify(analyzer, never()).sleepSeconds(anyInt());
186 
187                 // Except failure on delay due to bad value
188                 clientProvider.clearInvocations();
189                 settings.setInt(KEYS.ANALYZER_OSSINDEX_REQUEST_DELAY, 10);
190                 doNothing().when(analyzer).sleepSeconds(anyInt());
191                 assertThrows(AnalysisException.class, () -> analyzer.analyzeDependency(dependency, engine));
192                 verify(analyzer).sleepSeconds(10);
193             }
194         }

282     /*
283      * This action is inspired by the sprout method technique displayed in
284      * "Michael Feathers - Working Effectively with Legacy code".
285      *
286      * We want to trigger a race condition between a call to
287      * OssIndexAnalyzer.closeAnalyzer() and OssIndexAnalyzer.enrich().
288      *
289      * The last method access data from the "reports" field while
290      * closeAnalyzer() erase the reference. If enrich() is not included in
291      * the "FETCH_MUTEX" synchronized statement, we can trigger a
292      * NullPointerException in a multithreaded environment, which can happen
293      * due to the usage of java.util.concurrent.Future.
294      *
295      * We want to make sure enrich() will be able to set the url of an
296      * identifier and enrich it.
297      */
298     private static final class ClosedDuringEnrichOssIndexAnalyzer extends OssIndexAnalyzer {
299         private Future<?> pendingClosureTask;
300 
301         @Override
302         void enrich(Dependency dependency) {
303             @SuppressWarnings("resource") ExecutorService executor = Executors.newSingleThreadExecutor();
304             pendingClosureTask = executor.submit(() -> {
305                 try {
306                     this.closeAnalyzer();
307                 } catch (Exception e) {
308                     throw new RuntimeException(e);
309                 }
310             });
311             executor.shutdown();
312             super.enrich(dependency);
313         }
314 
315         void awaitPendingClosure() throws ExecutionException, InterruptedException {
316             pendingClosureTask.get();
317         }
318     }

320     private static final class SingleOkReportOssIndexClient implements OssindexClient {
321         @Override
322         public Map<PackageUrl, ComponentReport> requestComponentReports(List<PackageUrl> coordinates) throws Exception {
323             HashMap<PackageUrl, ComponentReport> reports = new HashMap<>();
324             ComponentReport report = requestComponentReport(coordinates.get(0));
325             reports.put(report.getCoordinates(), report);
326             return reports;
327         }
328 
329         @Override
330         public ComponentReport requestComponentReport(PackageUrl coordinates) throws Exception {
331             ComponentReport report = new ComponentReport();
332             report.setCoordinates(coordinates);
333             report.setReference(new URI("https://guide.sonatype.com/component/maven/test%3Atest/1.0"));
334             return report;
335         }
336 
337         @Override
338         public void close() {}
339     }

341     private static OssindexClient throwingOssIndex(Exception exception1) throws Exception {
342         OssindexClient client = mock(OssindexClient.class);
343         when(client.requestComponentReport(any())).thenThrow(exception1);
344         when(client.requestComponentReports(any())).thenThrow(exception1);
345         return client;
