81      @Nested
82      class BasePackagedSuppressionsLoading {
83          @Test
84          void packagedBaseSuppressions() throws Exception {
85              prepareBaseSuppressionsOnly();
86              @SuppressWarnings("unchecked") List<SuppressionRule> rules = (List<SuppressionRule>) engine.getObject(SUPPRESSION_OBJECT_KEY);
87              assertThat(rules, not(empty()));
88              assertThat("Expected all suppressions in base file to be marked as base", allRulesNotMarkedAsBase(rules), empty());
89          }
90      }

92      @Nested
93      class HostedSuppressionsLoading {
94          @Test
95          void packagedSnapshotHostedSuppressionsLoadedEvenIfRemoteHostedSuppressionsDisabled() throws Exception {
96              getSettings().setBoolean(KEYS.HOSTED_SUPPRESSIONS_ENABLED, false);
97              prepareHostedSuppressionsOnly();
98              @SuppressWarnings("unchecked") List<SuppressionRule> rules = (List<SuppressionRule>) engine.getObject(SUPPRESSION_OBJECT_KEY);
99              assertThat(rules, not(empty()));
100             assertThat("Expected all suppressions in hosted suppressions snapshot file to be marked as base", allRulesNotMarkedAsBase(rules), empty());
101         }

103         @Test
104         void packagedSnapshotHostedSuppressionsLoadedIfAutoUpdateDisabled() throws Exception {
105             getSettings().setBoolean(KEYS.HOSTED_SUPPRESSIONS_ENABLED, true);
106             getSettings().setBoolean(KEYS.AUTO_UPDATE, false);
107             prepareHostedSuppressionsOnly();
108             assertThat(AbstractSuppressionAnalyzer.getRuleCount(engine), greaterThan(0));

115             prepareHostedSuppressionsOnly();
116             assertThat(AbstractSuppressionAnalyzer.getRuleCount(engine), greaterThan(0));
117         }

156     @Nested
157     class UserSuppressionsLoading {
158         /**
159          * Test of getRules method, of class AbstractSuppressionAnalyzer for
160          * suppression file declared as URL.
161          */
162         @Test
163         void testGetRulesFromSuppressionFileFromURL() throws Exception {
164             final int numberOfExtraLoadedRules = getNumberOfRulesLoadedFromPath(testSuppressionsFileUrl()) - getNumberOfRulesLoadedInCoreFile();
165             assertEquals(TEST_SUPPRESSIONS_EXPECTED_COUNT, numberOfExtraLoadedRules, "Wrong # of expected extra user suppression rules");
166         }
167 
168         /**
169          * Test of getRules method, of class AbstractSuppressionAnalyzer for
170          * suppression file on the class path.
171          */
172         @Test
173         void testGetRulesFromSuppressionFileInClasspath() throws Exception {
174             final int numberOfExtraLoadedRules = getNumberOfRulesLoadedFromPath(TEST_SUPPRESSIONS_FILE) - getNumberOfRulesLoadedInCoreFile();
175             assertEquals(TEST_SUPPRESSIONS_EXPECTED_COUNT, numberOfExtraLoadedRules, "Wrong # of expected extra user suppression rules");
176         }
177 
178         /**
179          * Assert that rules are loaded from multiple files if multiple files are
180          * defined in the {@link Settings}.
181          */
182         @Test
183         void testGetRulesFromMultipleSuppressionFiles() throws Exception {
184             final int rulesInCoreFile = getNumberOfRulesLoadedInCoreFile();
185 
186             // GIVEN suppression rules from one file
187             final int rulesInFirstFile = getNumberOfRulesLoadedFromPath(TEST_SUPPRESSIONS_FILE) - rulesInCoreFile;
188 
189             // AND suppression rules from another file
190             final int rulesInSecondFile = getNumberOfRulesLoadedFromPath(OTHER_TEST_SUPPRESSIONS_FILE) - rulesInCoreFile;
191 
192             // WHEN initializing with both suppression files
193             final String[] suppressionFiles = {TEST_SUPPRESSIONS_FILE, OTHER_TEST_SUPPRESSIONS_FILE};
194             getSettings().setArrayIfNotEmpty(KEYS.SUPPRESSION_FILE, suppressionFiles);
195             prepareSuppressions();
196 
197             // THEN rules from both files were loaded
198             final int expectedSize = rulesInFirstFile + rulesInSecondFile + rulesInCoreFile;
199             assertThat("Expected suppressions from both files", AbstractSuppressionAnalyzer.getRuleCount(engine), is(expectedSize));
200         }
201 
202         @Test
203         void testFailureToLocateSuppressionFileAnywhere() {
204             getSettings().setString(Settings.KEYS.SUPPRESSION_FILE, "doesnotexist.xml");
205             assertThrows(InitializationException.class, AbstractSuppressionAnalyzerTest.this::prepareSuppressions);
206         }
207     }

209     /**
210      * Return the number of rules that are loaded from the core suppression
211      * file.
212      *
213      * @return the number of rules defined in the core suppression file.
214      * @throws Exception if loading the rules fails.
215      */
216     private int getNumberOfRulesLoadedInCoreFile() throws Exception {
217         getSettings().removeProperty(KEYS.SUPPRESSION_FILE);
218         prepareSuppressions();
219         return AbstractSuppressionAnalyzer.getRuleCount(engine);
220     }
221 
222     /**
223      * Load a file into the {@link AbstractSuppressionAnalyzer} and return the
224      * number of rules loaded.
225      *
226      * @param path the path to load.
227      * @return the number of rules that were loaded (including the core rules).
228      * @throws Exception if loading the rules fails.
229      */
230     private int getNumberOfRulesLoadedFromPath(final String path) throws Exception {
231         getSettings().setString(KEYS.SUPPRESSION_FILE, path);
232         prepareSuppressions();
233         return AbstractSuppressionAnalyzer.getRuleCount(engine);
234     }
235 
236     private void prepareSuppressions() throws InvalidSettingException, InitializationException {
237         engine = new Engine(Mode.EVIDENCE_COLLECTION, getSettings());
238         newAnalyzer().prepare(engine);
239     }
240 
241     private void prepareBaseSuppressionsOnly() throws InvalidSettingException, SuppressionParseException {
242         engine = new Engine(Mode.EVIDENCE_COLLECTION, getSettings());
243         newAnalyzer().loadPackagedBaseSuppressionData(engine);
244     }
245 
246     private void prepareHostedSuppressionsOnly() throws InvalidSettingException {
247         engine = new Engine(Mode.EVIDENCE_COLLECTION, getSettings());
248         newAnalyzer().loadHostedSuppressionBaseData(engine);
249     }
250 
251     private @NonNull AbstractSuppressionAnalyzerImpl newAnalyzer() throws InvalidSettingException {
252         final AbstractSuppressionAnalyzerImpl fileAnalyzer = new AbstractSuppressionAnalyzerImpl();
253         fileAnalyzer.initialize(getSettings());
254         Downloader.getInstance().configure(getSettings());
255         return fileAnalyzer;
256     }
257 
258     private @NonNull List<SuppressionRule> allRulesNotMarkedAsBase(List<SuppressionRule> baseRules) {
259         return baseRules.stream().filter(r -> !r.isBase()).collect(Collectors.toList());
260     }
