46  class RetireJSDataSourceTest extends BaseTest {
47      private static final String TEST_RETIRE_JS_REPOSITORY_FILE = "retirejs/jsrepository.json";
48  
49      private RetireJSDataSource dataSource;
50      private Engine engine;
51  
52      @BeforeEach
53      public void createEngine() {
54          dataSource = new RetireJSDataSource();
55          engine = new Engine(Engine.Mode.EVIDENCE_COLLECTION, getSettings());
56      }
57  
58      @AfterEach
59      void closeEngine() {
60          if (engine != null) {
61              dataSource.purge(engine);
62              engine.close();
63          }
64      }
65  
66      @Nested
67      class Update {
68          @Test
69          void doesNothingIfRetireJsDisabled() throws Exception {
70              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, false);
71              dataSource.update(engine);
72              assertNoCachedRetireJs();
73          }
74  
75          @Test
76          void loadsRemoteIfEmptyWithAutoUpdateDisabled() throws Exception {
77              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
78              getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, false);
79              dataSource.update(engine);
80              assertCachedRetireJs();
81          }
82  
83          @Test
84          void doesNothingIfPresentAndAutoUpdateDisabled() throws Exception {
85              dataSource.update(engine);
86  
87              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
88              getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, false);
89              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_FORCEUPDATE, false);
90  
91              // try again with repo present; but with a URL that would fail if invoked
92              getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl().replace(TEST_RETIRE_JS_REPOSITORY_FILE, "doesnt-exist/" + TEST_RETIRE_JS_REPOSITORY_FILE));
93              dataSource.update(engine);
94          }
95  
96          @Test
97          void failsIfRemoteRetireJsRepoNotFound() throws Exception {
98              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
99              getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
100             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, "file:///does-not-exist.xml");
101             var ex = assertThrows(UpdateException.class, () -> dataSource.update(engine));
102             assertThat(ex.getMessage(), containsString("Failed to initialize the RetireJS repo"));
103             assertThat(ExceptionUtils.getRootCause(ex), instanceOf(NoSuchFileException.class));
104             assertNoCachedRetireJs();
105         }
106 
107         @Test
108         void failsIfUrlDoesntIncludeAFileName() {
109             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
110             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, "https://valid.url.but.no.file/");
111             var ex = assertThrowsExactly(UpdateException.class, () -> dataSource.update(engine));
112             assertThat(ex.getMessage(), containsString("Unable to determine the local location to cache"));
113             assertThat(ex.getCause(), instanceOf(InvalidSettingException.class));
114             assertThat(ex.getCause().getMessage(), containsString("RetireJS URL must imply a filename"));
115         }
116 
117         @Test
118         void failsIfUrlIsMalformed() {
119             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
120             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, "bad-url");
121             var ex = assertThrowsExactly(UpdateException.class, () -> dataSource.update(engine));
122             assertThat(ex.getMessage(), containsString("Invalid URL for RetireJS repository (bad-url)"));
123             assertThat(ex.getCause(), instanceOf(MalformedURLException.class));
124         }
125 
126         @Test
127         void loadsRemoteRetireJsIfEnabledAndForced() throws Exception {
128             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
129             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, false);
130             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_FORCEUPDATE, true);
131             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
132             dataSource.update(engine);
133 
134             // try again with repo present; but with a URL that fails when invoked
135             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl().replace(TEST_RETIRE_JS_REPOSITORY_FILE, "doesnt-exist/" + TEST_RETIRE_JS_REPOSITORY_FILE));
136             var ex = assertThrowsExactly(UpdateException.class, () -> dataSource.update(engine));
137             assertThat(ExceptionUtils.getRootCause(ex), instanceOf(NoSuchFileException.class));
138         }
139 
140         @Test
141         void loadsRemoteRetireJsIfEnabledWithAutoUpdate() throws Exception {
142             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
143             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
144             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
145             dataSource.update(engine);
146             assertThat(Files.readString(cachedRepoFile()), is(testRepositoryFileContent()));
147             assertThat(Files.exists(cachedRepoFileProperties()), is(true));
148         }
149 
150         @Test
151         void doesNothingIfRemoteRetireJsIsNotStale() throws Exception {
152             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
153             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
154             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
155             getSettings().setInt(Settings.KEYS.ANALYZER_RETIREJS_REPO_VALID_FOR_HOURS, 1);
156             dataSource.update(engine);
157 
158             // Update again immediately
159             String firstUpdateProperties = Files.readString(cachedRepoFileProperties());
160             FileTime firstUpdatePropertiesModified = Files.getLastModifiedTime(cachedRepoFileProperties());
161             dataSource.update(engine);
162 
163             assertThat(Files.readString(cachedRepoFileProperties()), is(firstUpdateProperties));
164             assertThat(Files.getLastModifiedTime(cachedRepoFileProperties()), is(firstUpdatePropertiesModified));
165         }
166 
167         @Test
168         void reloadsRemoteRetireJsIfStale() throws Exception {
169             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
170             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
171             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
172             getSettings().setInt(Settings.KEYS.ANALYZER_RETIREJS_REPO_VALID_FOR_HOURS, 0);
173             dataSource.update(engine);
174 
175             // Reset to force an update
176             Files.writeString(cachedRepoFile(), "stale content");
177 
178             dataSource.update(engine);
179             assertThat(Files.readString(cachedRepoFile()), not("stale content"));
180         }
181     }
182 
183     @Nested
184     class Purge {
185         @Test
186         void purgeRemovesCachedFiles() throws Exception {
187             getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
188             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
189             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
190             dataSource.update(engine);
191 
192             assertThat(Files.exists(cachedRepoFile()), is(true));
193             dataSource.purge(engine);
194             assertThat(Files.exists(cachedRepoFile()), is(false));
195         }
196 
197         @Test
198         void doesNothingIfNoCachedFile() throws Exception {
199             dataSource.purge(engine);
200             assertThat(Files.exists(cachedRepoFile()), is(false));
201         }
202 
203         @Test
204         void doesNothingIfUrlIsMalformed() {
205             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, "bad-url");
206             dataSource.purge(engine);
207         }
208     }
209 
210     private @NonNull Path cachedRepoFile() throws UpdateException {
211         return dataSource.validatedRepoFile().toPath();
212     }
213 
214     private @NonNull Path cachedRepoFileProperties() throws UpdateException {
215         return Path.of(cachedRepoFile() + ".properties");
216     }
217 
218     private @NonNull String testRepositoryFileUrl() {
219         return BaseTest.getResourceAsUrlString(this, TEST_RETIRE_JS_REPOSITORY_FILE);
220     }
221 
222     private @NonNull String testRepositoryFileContent() {
223         return BaseTest.getResourceAsContentString(this, TEST_RETIRE_JS_REPOSITORY_FILE);
224     }
225 
226     private void assertNoCachedRetireJs() throws UpdateException {
227         assertThat("RetireJS repo file should not exist", dataSource.validatedRepoFile().exists(), is(false));
228     }
229 
230     private void assertCachedRetireJs() throws UpdateException {
231         assertThat("RetireJS repo file should exist", dataSource.validatedRepoFile().exists(), is(true));
232     }
233 }
