20  import org.apache.commons.lang3.exception.ExceptionUtils;
21  import org.json.JSONException;
22  import org.jspecify.annotations.NonNull;
23  import org.junit.jupiter.api.AfterEach;
24  import org.junit.jupiter.api.BeforeEach;
25  import org.junit.jupiter.api.Nested;
26  import org.junit.jupiter.api.Test;
27  import org.owasp.dependencycheck.BaseTest;
28  import org.owasp.dependencycheck.Engine;
29  import org.owasp.dependencycheck.data.update.RetireJSDataSource;
30  import org.owasp.dependencycheck.exception.InitializationException;
31  import org.owasp.dependencycheck.utils.Downloader;
32  import org.owasp.dependencycheck.utils.InvalidSettingException;
33  import org.owasp.dependencycheck.utils.Settings;
34  
35  import java.nio.file.NoSuchFileException;
36  
37  import static org.hamcrest.CoreMatchers.is;
38  import static org.hamcrest.MatcherAssert.assertThat;
39  import static org.hamcrest.Matchers.containsString;
40  import static org.hamcrest.Matchers.instanceOf;
41  import static org.junit.jupiter.api.Assertions.assertThrows;
42  
43  class RetireJsAnalyzerTest extends BaseTest {
44      private static final String TEST_RETIRE_JS_REPOSITORY_FILE = "retirejs/jsrepository.json";
45  
46      private RetireJsAnalyzer analyzer;
47      private Engine engine;
48  
49      @BeforeEach
50      void settings() {
51          getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_ENABLED, true);
52          getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_FORCEUPDATE, false);
53          getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
54      }
55      
56      @AfterEach
57      void cleanUp() {
58          if (engine != null) {
59              new RetireJSDataSource().purge(engine);
60              engine.close();
61          }
62      }
63  
64      private String testRepositoryFileUrl() {
65          return BaseTest.getResourceAsUrlString(this, TEST_RETIRE_JS_REPOSITORY_FILE);
66      }
67  
68      @Nested
69      class RepositoryLoading {
70  
71          @Test
72          void loadsRemoteIfEmptyWithAutoUpdateDisabled() throws Exception {
73              getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, false);
74              prepareRetireJs();
75              assertThat(analyzer.knownLibraryCountFor("retire-example-0.0.1.js").orElseThrow(), is(1));
76          }
77  
78          @Test
79          void loadsRemoteIfPresentAndForced() throws Exception {
80              prepareRetireJs();
81  
82              // try again with repo present; but with a URL that will fail when invoked so we know it tried
83              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_FORCEUPDATE, true);
84              getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl().replace(TEST_RETIRE_JS_REPOSITORY_FILE, "doesnt-exist/" + TEST_RETIRE_JS_REPOSITORY_FILE));
85  
86              var ex = assertThrows(InitializationException.class, RetireJsAnalyzerTest.this::prepareRetireJs);
87              assertThat(ex.getMessage(), containsString("Failed to initialize the RetireJS repo"));
88              assertThat(ExceptionUtils.getRootCause(ex), instanceOf(NoSuchFileException.class));
89          }
90  
91          @Test
92          void doesNothingIfPresentAndAutoUpdateDisabled() throws Exception {
93              prepareRetireJs();
94  
95              getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, false);
96              getSettings().setBoolean(Settings.KEYS.ANALYZER_RETIREJS_FORCEUPDATE, false);
97  
98              // try again with repo present; but with a URL that would fail if invoked
99              getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl().replace(TEST_RETIRE_JS_REPOSITORY_FILE, "doesnt-exist/" + TEST_RETIRE_JS_REPOSITORY_FILE));
100             prepareRetireJs();
101         }
102 
103         @Test
104         void failsIfRemoteRetireJsRepoNotFound() {
105             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
106             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, "file:///does-not-exist.xml");
107             var ex = assertThrows(InitializationException.class, RetireJsAnalyzerTest.this::prepareRetireJs);
108             assertThat(ex.getMessage(), containsString("Failed to initialize the RetireJS repo"));
109             assertThat(ExceptionUtils.getRootCause(ex), instanceOf(NoSuchFileException.class));
110             assertThat(analyzer.isEnabled(), is(false));
111         }
112 
113         @Test
114         void failsIfRemoteRetireJsCannotBeParsed() {
115             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, BaseTest.getResourceAsUrlString(this, "retirejs/jsrepository-invalid.json"));
116             var ex = assertThrows(InitializationException.class, RetireJsAnalyzerTest.this::prepareRetireJs);
117             assertThat(ex.getMessage(), containsString("Failed to initialize the RetireJS repo"));
118             assertThat(ExceptionUtils.getRootCause(ex), instanceOf(JSONException.class));
119             assertThat(analyzer.isEnabled(), is(false));
120         }
121 
122         @Test
123         void prefersRemoteRetireJsIfEnabled() throws Exception {
124             getSettings().setBoolean(Settings.KEYS.AUTO_UPDATE, true);
125             getSettings().setString(Settings.KEYS.ANALYZER_RETIREJS_REPO_JS_URL, testRepositoryFileUrl());
126             prepareRetireJs();
127             assertThat(analyzer.knownLibraryCountFor("retire-example-0.0.1.js").orElseThrow(), is(1));
128         }
129     }
130 
131     private void prepareRetireJs() throws InvalidSettingException, InitializationException {
132         engine = new Engine(Engine.Mode.EVIDENCE_COLLECTION, getSettings());
133         analyzer = newAnalyzer();
134         analyzer.prepareFileTypeAnalyzer(engine);
135     }
136 
137     private @NonNull RetireJsAnalyzer newAnalyzer() throws InvalidSettingException {
138         final RetireJsAnalyzer fileAnalyzer = new RetireJsAnalyzer();
139         fileAnalyzer.initialize(getSettings());
140         Downloader.getInstance().configure(getSettings());
141         return fileAnalyzer;
142     }
143 }
