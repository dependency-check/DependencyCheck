80                  LOGGER.warn("Exception setting up AssemblyAnalyzer. Tests will be incomplete");

82                  LOGGER.warn("Exception setting up AssemblyAnalyzer. Tests will be incomplete");
83              }
84              assumeTrue(false, "Is dotnet installed? TESTS WILL BE INCOMPLETE: " + e);
85          }
86      }

88      private void assertGrokAssembly() {
89          // There must be an .exe and a .config files created in the temp
90          // directory and they must match the resources they were created from.
91          File grokAssemblyExeFile = analyzer.getGrokAssemblyPath();
92          assertTrue(grokAssemblyExeFile.isFile(), "The GrokAssembly executable was not created.");

103     @Test
104     void testAnalysis() throws Exception {
105         assumeTrue(analyzer.buildArgumentList() != null);
106         File f = analyzer.getGrokAssemblyPath();
107         Dependency d = new Dependency(f);
108         analyzer.analyze(d, null);
109         assertTrue(d.contains(EvidenceType.VENDOR, new Evidence("grokassembly", "CompanyName", "OWASP Contributors", Confidence.HIGHEST)));
110         assertTrue(d.contains(EvidenceType.PRODUCT, new Evidence("grokassembly", "ProductName", "GrokAssembly", Confidence.HIGHEST)));
111     }

128     void testNonexistent() {
129         assumeTrue(analyzer.buildArgumentList() != null);
130 
131         // Tweak the log level so the warning doesn't show in the console
132         String oldProp = System.getProperty(LOG_KEY, "info");
133         File f = BaseTest.getResourceAsFile(this, "log4net.dll");
134         File test = new File(f.getParent(), "nonexistent.dll");
135         Dependency d = new Dependency(test);
136 
137         try {
138             AnalysisException ae = assertThrows(AnalysisException.class, () -> analyzer.analyze(d, null),
139                 "Expected an AnalysisException");
140             assertTrue(ae.getMessage().contains("nonexistent.dll does not exist and cannot be analyzed by dependency-check"));
141         } finally {
142             System.setProperty(LOG_KEY, oldProp);
143         }
144     }

146     @Test
147     void testWithSettingMono() {
148 
149         //This test doesn't work on Windows.
150         assumeFalse(System.getProperty("os.name").startsWith("Windows"));
151 
152         String oldValue = getSettings().getString(Settings.KEYS.ANALYZER_ASSEMBLY_DOTNET_PATH);
153         // if oldValue is null, that means that neither the system property nor the setting has
154         // been set. If that's the case, then we have to make it such that when we recover,
155         // null still comes back. But you can't put a null value in a HashMap, so we have to set
156         // the system property rather than the setting.
157         System.setProperty(Settings.KEYS.ANALYZER_ASSEMBLY_DOTNET_PATH, "/yooser/bine/mono");
158 
159         String oldProp = System.getProperty(LOG_KEY, "info");
160         try {
161             // Tweak the logging to swallow the warning when testing
162             System.setProperty(LOG_KEY, "error");
163             // Have to make a NEW analyzer because during setUp, it would have gotten the correct one
164             AssemblyAnalyzer aanalyzer = new AssemblyAnalyzer();
165             aanalyzer.initialize(getSettings());
166             aanalyzer.accept(new File("test.dll")); // trick into "thinking it is active"
167 
168             InitializationException ae = assertThrows(InitializationException.class, () -> aanalyzer.prepare(null),
169                     "Expected an InitializationException");
170             assertEquals("An error occurred with the .NET AssemblyAnalyzer, is the dotnet 8.0 runtime or sdk installed?", ae.getMessage());
171         } finally {
172             System.setProperty(LOG_KEY, oldProp);
173             // Recover the logger
174             // Now recover the way we came in. If we had to set a System property, delete it. Otherwise,
175             // reset the old value
176             if (oldValue == null) {
177                 System.getProperties().remove(Settings.KEYS.ANALYZER_ASSEMBLY_DOTNET_PATH);
178             } else {
179                 System.setProperty(Settings.KEYS.ANALYZER_ASSEMBLY_DOTNET_PATH, oldValue);
180             }
181         }
182     }

184     @Test
185     void testAzureIdentity() throws MalformedPackageURLException {
186         // Given
187         var data = newAzureIdentityAssemblyData();
188         var dependency = newAzureIdentityDependency();
189 
190         var expectedDescription = "Microsoft Azure.Identity Component\n\nThis is the implementation of the Azure SDK " +
191                 "Client Library for Azure Identity";
192         var expectedVersionEvidences = expectedAzureIdentityVersionEvidences();
193         var expectedVersion = "1.7.0";
194         var expectedProductEvidences = expectedAzureIdentityProductEvidences();
195         var expectedVendorEvidences = expectedAzureIdentityVendorEvidences();
196         var expectedName = "Azure.Identity";
197         var expectedIdentifiers = Set.of(new PurlIdentifier(new PackageURL("pkg:generic/Azure.Identity@1.7.0"),
198                 Confidence.MEDIUM));
199         var expectedEcosystem = "dotnet";
200 
201         // When
202         analyzer.updateDependency(data, dependency);
203 
204         // Then
205         assertEquals(expectedDescription, dependency.getDescription());
206         assertEquals(expectedVersionEvidences, dependency.getEvidence(EvidenceType.VERSION));
207         assertEquals(expectedVersion, dependency.getVersion());
208         assertEquals(expectedProductEvidences, dependency.getEvidence(EvidenceType.PRODUCT));
209         assertEquals(expectedVendorEvidences, dependency.getEvidence(EvidenceType.VENDOR));
210         assertEquals(expectedName, dependency.getName());
211         assertEquals(expectedIdentifiers, dependency.getSoftwareIdentifiers());
212         assertEquals(expectedEcosystem, dependency.getEcosystem());
213     }
