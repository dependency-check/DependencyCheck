25  import org.owasp.dependencycheck.BaseTest;

30  import java.io.File;
31  
32  import static org.junit.jupiter.api.Assertions.assertEquals;
33  import static org.junit.jupiter.api.Assertions.assertFalse;
34  import static org.junit.jupiter.api.Assertions.assertTrue;
35  
36  /**
37   * Unit tests for {@link PinnedMavenInstallAnalyzer}.
38   */
39  class PinnedMavenInstallAnalyzerTest extends BaseDBTestCase {
40  
41      /**
42       * The analyzer to test.
43       */
44      private PinnedMavenInstallAnalyzer analyzer;
45  
46      /**
47       * Correctly set up the analyzer for testing.
48       *
49       * @throws Exception thrown if there is a problem
50       */
51      @BeforeEach
52      @Override
53      public void setUp() throws Exception {
54          super.setUp();
55          analyzer = new PinnedMavenInstallAnalyzer();
56          analyzer.initialize(getSettings());
57          analyzer.setFilesMatched(true);
58          analyzer.prepare(null);
59      }
60  
61      /**
62       * Cleanup the analyzer's temp files, etc.
63       *
64       * @throws Exception thrown if there is a problem
65       */
66      @AfterEach
67      @Override
68      public void tearDown() throws Exception {
69          analyzer.close();
70          super.tearDown();
71      }
72  
73      @Test
74      void testGetName() {
75          assertEquals("Pinned Maven install Analyzer", analyzer.getName());
76      }
77  
78      @Test
79      void testSupportsFiles() {
80          assertTrue(analyzer.accept(new File("install_maven.json")));
81          assertTrue(analyzer.accept(new File("maven_install.json")));
82          assertTrue(analyzer.accept(new File("maven_install_v010.json")));
83          assertTrue(analyzer.accept(new File("maven_install_v2.json")));
84          assertTrue(analyzer.accept(new File("rules_jvm_external_install.json")));
85          assertTrue(analyzer.accept(new File("pinned_install_gplonly.json")));
86          assertFalse(analyzer.accept(new File("install.json")), "should not accept Cloudflare install.json");
87          assertFalse(analyzer.accept(new File("maven_install.txt")));
88          assertFalse(analyzer.accept(new File("pinned.json")));
89          assertFalse(analyzer.accept(new File("install.json.zip")));
90      }
91  
92      /**
93       * Tests that the analyzer correctly pulls dependencies out of a pinned v0.1.0 {@code maven_install.json}.
94       */
95      @Test
96      void testAnalyzePinnedInstallJsonV010() throws Exception {
97          try (Engine engine = new Engine(getSettings())) {
98              final Dependency result = new Dependency(BaseTest.getResourceAsFile(this, "maven_install_v010.json"));
99              engine.addDependency(result);
100             analyzer.analyze(result, engine);
101             assertFalse(ArrayUtils.contains(engine.getDependencies(), result));
102             assertEquals(10, engine.getDependencies().length);
103             boolean found = false;
104             for (Dependency d : engine.getDependencies()) {
105                 if ("com.google.errorprone:error_prone_annotations".equals(d.getName())) {
106                     found = true;
107                     assertEquals("2.3.4", d.getVersion());
108                     assertEquals(Ecosystem.JAVA, d.getEcosystem());
109                 }
110             }
111             assertTrue(found, "Expected to find com.google.errorprone:error_prone_annotations:2.3.4");
112         }
113     }
114 
115     /**
116      * Tests that the analyzer correctly pulls dependencies out of a pinned v2 {@code maven_install.json}.
117      */
118     @Test
119     void testAnalyzePinnedInstallJsonV2() throws Exception {
120         try (Engine engine = new Engine(getSettings())) {
121             final Dependency result = new Dependency(BaseTest.getResourceAsFile(this, "maven_install_v2.json"));
122             engine.addDependency(result);
123             analyzer.analyze(result, engine);
124             assertFalse(ArrayUtils.contains(engine.getDependencies(), result));
125             assertEquals(113, engine.getDependencies().length);
126             boolean found = false;
127             for (Dependency d : engine.getDependencies()) {
128                 if ("io.grpc:grpc-protobuf".equals(d.getName())) {
129                     found = true;
130                     assertEquals("1.48.1", d.getVersion());
131                     assertEquals(Ecosystem.JAVA, d.getEcosystem());
132                 }
133             }
134             assertTrue(found, "Expected to find com.google.errorprone:error_prone_annotations:2.3.4");
135         }
136     }
137 
138     /**
139      * Tests that the analyzer ignores a Cloudflare-style {@code install.json}.
140      */
141     @Test
142     void testAnalyzeOtherInstallJson() throws Exception {
143         try (Engine engine = new Engine(getSettings())) {
144             final Dependency result = new Dependency(BaseTest.getResourceAsFile(this, "install.json"));
145             engine.addDependency(result);
146             analyzer.analyze(result, engine);
147             assertTrue(ArrayUtils.contains(engine.getDependencies(), result));
148             assertEquals(1, engine.getDependencies().length);
149         }
150     }
