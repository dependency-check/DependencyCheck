113     /**
114      * If a suppression is present in a group and does not have attributes set, then the ones from the group are used
115      * as defaults.
116      */
117     @Test
118     void testParseSuppressionV1dot4Inherits() throws SuppressionParseException {
119         File file = BaseTest.getResourceAsFile(this, "suppressions_1_4.xml");
120         SuppressionParser instance = new SuppressionParser();
121         List<SuppressionRule> suppressionRules = instance.parseSuppressionRules(file);
122 
123         // CVE-2013-1338 in test xml has no attributes and should inherit the ones set on group level.
124         List<SuppressionRule> filteredSuppressions = suppressionRules.stream().
125                 filter(s -> s.getCve().contains("CVE-2013-1338"))
126                 .collect(Collectors.toList());
127         assertEquals(1, filteredSuppressions.size());
128         SuppressionRule rule = filteredSuppressions.get(0);
129 
130         Instant expectedTime = LocalDate.of(2046, 1, 1)
131                 .atStartOfDay(ZoneOffset.UTC)
132                 .toInstant();
133         assertEquals(expectedTime, rule.getUntil().toInstant());
134         assertTrue(rule.isBase());
135     }
136 
137     /**
138      * If a suppression in a suppression group has attributes set, then those override those of the suppressionGroup.
139      */
140     @Test
141     void testParseSuppressionV1dot4AttributeOverrides() throws SuppressionParseException {
142         File file = BaseTest.getResourceAsFile(this, "suppressions_1_4.xml");
143         SuppressionParser instance = new SuppressionParser();
144         List<SuppressionRule> suppressionRules = instance.parseSuppressionRules(file);
145 
146         // CVE-2013-1339 in test xml has attribute {code (until="2027-01-01Z")} set and is present in  suppressionGroup.
147         List<SuppressionRule> filteredSuppressions = suppressionRules.stream().
148                 filter(s -> s.getCve().contains("CVE-2013-1339"))
149                 .collect(Collectors.toList());
150         assertEquals(1, filteredSuppressions.size());
151         SuppressionRule rule = filteredSuppressions.get(0);
152 
153         Instant expectedTime = LocalDate.of(2027, 1, 1)
154                 .atStartOfDay(ZoneOffset.UTC)
155                 .toInstant();
156         assertEquals(expectedTime, rule.getUntil().toInstant());
157         assertFalse(rule.isBase());
158     }
