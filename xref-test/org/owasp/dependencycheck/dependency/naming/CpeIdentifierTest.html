15      @Test
16      void testNvdSearchUrlFormatting() throws CpeValidationException {
17          String encodedUrl = CpeIdentifier.nvdSearchUrlFor(new CpeBuilder().part(Part.APPLICATION).vendor("apache").product("struts").version("1.9.0").build());
18          assertEquals("https://nvd.nist.gov/vuln/search#/nvd/home?sortOrder=3&sortDirection=2&cpeFilterMode=applicability&resultType=records&" +
19                  "cpeName=cpe%3A2.3%3Aa%3Aapache%3Astruts%3A1.9.0%3A%2A%3A%2A%3A%2A%3A%2A%3A%2A%3A%2A%3A%2A", encodedUrl);
20          assertEquals("https://nvd.nist.gov/vuln/search#/nvd/home?sortOrder=3&sortDirection=2&cpeFilterMode=applicability&resultType=records&" +
21                  "cpeName=cpe:2.3:a:apache:struts:1.9.0:*:*:*:*:*:*:*", URLDecoder.decode(encodedUrl, UTF_8));
22      }
