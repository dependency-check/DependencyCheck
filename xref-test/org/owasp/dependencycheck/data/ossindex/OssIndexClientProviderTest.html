33      private final Settings settings = new Settings();
34  
35      @Captor
36      ArgumentCaptor<OssindexClientConfiguration> configCaptor;
37      @Captor
38      ArgumentCaptor<Transport> transportCaptor;
39      @Captor
40      ArgumentCaptor<Marshaller> marshallerCaptor;

42      @BeforeEach
43      void setUp() {
44          setSonatypeGuideCredentials(settings);

47      @Test
48      void shouldUseDefaultConfiguration() {
49          try (var clientCreation = OssIndexHelper.stubClientCreation()) {
50              OssIndexClientProvider.create(settings);
51              clientCreation.verify(() -> OssIndexClientProvider.newClientFor(configCaptor.capture(), transportCaptor.capture(), marshallerCaptor.capture()));
52  
53              OssindexClientConfiguration config = configCaptor.getValue();
54              assertThat(config.getAuthConfiguration().getPassword(), is("sonatype_pat_abcdef"));
55              assertThat(config.getBaseUrl(), is(URI.create(OssIndexClientProvider.DEFAULT_BASE_URL)));
56              assertThat(config.getBatchSize(), is(128));
57  
58              ODCConnectionTransport transport = assertTransportConfigured();
59              assertThat(transport.getUserAgent().get(), matchesPattern("dependency-check/.+ \\(.*\\)"));
60  
61              assertThat(marshallerCaptor.getValue(), instanceOf(GsonMarshaller.class));
62          }
63      }
64  
65      @Test
66      void canChangeConfigValues() {
67          settings.setString(Settings.KEYS.ANALYZER_OSSINDEX_URL, "https://some.other.url");
68          settings.setInt(Settings.KEYS.ANALYZER_OSSINDEX_BATCH_SIZE, 100);
69  
70          try (var clientCreation = OssIndexHelper.stubClientCreation()) {
71              OssIndexClientProvider.create(settings);
72              clientCreation.verify(() -> OssIndexClientProvider.newClientFor(configCaptor.capture(), transportCaptor.capture(), marshallerCaptor.capture()));
73  
74              ODCConnectionTransport transport = assertTransportConfigured();
75              assertThat(transport.getUserAgent().get(), matchesPattern("dependency-check/.+ \\(.*\\)"));
76  
77              OssindexClientConfiguration config = configCaptor.getValue();
78              assertThat(config.getBaseUrl(), is(URI.create("https://some.other.url")));
79              assertThat(config.getBatchSize(), is(100));
80          }
81      }
82  
83      private @NonNull ODCConnectionTransport assertTransportConfigured() {
84          Transport transport = transportCaptor.getValue();
85          assertThat(transport, instanceOf(ODCConnectionTransport.class));
86          return (ODCConnectionTransport) transport;
87      }

90      class CacheConfig {
91          @Test
92          void shouldUseDefaultCacheConfiguration() throws IOException {
93              try (var clientCreation = OssIndexHelper.stubClientCreation()) {
94                  OssIndexClientProvider.create(settings);
95                  clientCreation.verify(() -> OssIndexClientProvider.newClientFor(configCaptor.capture(), transportCaptor.capture(), marshallerCaptor.capture()));
96  
97                  DirectoryCache.Configuration config = assertDirectoryCacheConfigured();

105             settings.setInt(Settings.KEYS.ANALYZER_OSSINDEX_CACHE_VALID_FOR_HOURS, 1);
106             try (var clientCreation = OssIndexHelper.stubClientCreation()) {
107                 OssIndexClientProvider.create(settings);
108                 clientCreation.verify(() -> OssIndexClientProvider.newClientFor(configCaptor.capture(), transportCaptor.capture(), marshallerCaptor.capture()));
109 
110                 DirectoryCache.Configuration config = assertDirectoryCacheConfigured();
111                 assertThat(config.getExpireAfter(), is(Duration.standardHours(1)));
112             }
113         }

115         private DirectoryCache.Configuration assertDirectoryCacheConfigured() {
116             OssindexClientConfiguration config = configCaptor.getValue();
