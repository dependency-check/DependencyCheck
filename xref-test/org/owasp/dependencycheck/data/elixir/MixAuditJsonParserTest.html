25      void testSingleResult() throws Exception {
26          try (Reader reader = new InputStreamReader(BaseTest.getResourceAsStream(this, "elixir/mix_audit/plug.json"), UTF_8)) {
27              MixAuditJsonParser parser = new MixAuditJsonParser(reader);
28              parser.process();
29  
30              assertEquals(1, parser.getResults().size(), "must have 1 result");
31  
32              MixAuditResult r = parser.getResults().get(0);
33              assertEquals("dc96aba4-4462-4d3b-b73f-28b9349133d8", r.getId());
34              assertEquals("2018-1000883", r.getCve());
35              assertEquals("Header Injection\n", r.getTitle());
36              assertEquals("Cookie headers were not validated\n", r.getDescription());
37              assertEquals("https://github.com/elixir-plug/plug/commit/8857f8ab4acf9b9c22e80480dae2636692f5f573", r.getUrl());
38              assertEquals("/DependencyCheck/core/src/test/resources/elixir/vulnerable/mix.lock", r.getDependencyLockfile());

40              assertEquals("1.3.4", r.getDependencyVersion());
