24  class PluginLoaderTest {
25  
26      private final Instrumentation instrumentation = Mockito.mock(Instrumentation.class);
27  
28      @TempDir
29      Path tempDir;
30  
31      @Test
32      void shouldDoNothingIfDirectoryDoesntExist() {
33          PluginLoader.premain("blah", instrumentation);
34          verifyNoInteractions(instrumentation);
35      }
36  
37      @Test
38      void shouldDoNothingIfDirectoryIsEmpty() {
39          PluginLoader.premain(tempDir.toString(), instrumentation);
40          verifyNoInteractions(instrumentation);
41      }
42  
43      @Test
44      void shouldAddJarToClassPath() throws Exception {
45          createEmptyValidJar();
46          createEmptyValidJar();
47          PluginLoader.premain(tempDir.toString(), instrumentation);
48          verify(instrumentation, times(2))
49                  .appendToSystemClassLoaderSearch(assertArg(jar -> assertThat(jar.getName(), matchesPattern(".*/dummy.*\\.jar"))));
50      }
51  
52      @Test
53      void shouldStopLoadingPluginsOnBadJarButSucceed() throws Exception {
54          PrintStream originalErr = System.err;
55          ByteArrayOutputStream errContent = new ByteArrayOutputStream();
56          System.setErr(new PrintStream(errContent, true, UTF_8));
57          try {
58              createEmptyBadJar();
59              PluginLoader.premain(tempDir.toString(), instrumentation);
60              assertThat(errContent.toString(), matchesPattern(Pattern.compile("\\[WARN] Failed to read plugin jar file at .*/dummy.*\\.jar\\. Jar will not be available on classpath.*zip file is empty.*", Pattern.DOTALL)));
61          } finally {
62              System.setErr(originalErr);
63          }
64      }
65  
66      private Path createEmptyBadJar() throws IOException {
67          return Files.createTempFile(tempDir, "dummy", ".jar");
68      }
69  
70      private void createEmptyValidJar() throws IOException {
71          new ZipArchiveOutputStream(createEmptyBadJar().toFile()).close();
72      }
73  }
